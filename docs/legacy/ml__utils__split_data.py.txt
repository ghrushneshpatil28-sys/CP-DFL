import os
import shutil
from sklearn.model_selection import train_test_split

BASE_DIR = r"d:\COLLEGE\SEM-4\EDI-2\New folder\lung_cancer_detection\backend\DataSet"
POOL_DIR = os.path.join(BASE_DIR, "pool")
TRAIN_DIR = os.path.join(BASE_DIR, "train")
TEST_DIR = os.path.join(BASE_DIR, "test")

def create_splits():
    # Ensure fresh dirs
    for d in [TRAIN_DIR, TEST_DIR]:
        if os.path.exists(d):
            shutil.rmtree(d, ignore_errors=True)
        os.makedirs(d)
        
    classes = os.listdir(POOL_DIR)
    
    for cls in classes:
        os.makedirs(os.path.join(TRAIN_DIR, cls))
        os.makedirs(os.path.join(TEST_DIR, cls))
        
        class_files = os.listdir(os.path.join(POOL_DIR, cls))
        if len(class_files) < 10:
            print(f"[WARN] Not enough files in {cls} to split.")
            continue
            
        train_files, test_files = train_test_split(class_files, test_size=0.15, random_state=42)
        
        for f in train_files:
            shutil.copy(os.path.join(POOL_DIR, cls, f), os.path.join(TRAIN_DIR, cls, f))
            
        for f in test_files:
            shutil.copy(os.path.join(POOL_DIR, cls, f), os.path.join(TEST_DIR, cls, f))
            
        print(f"{cls}: Train={len(train_files)}, Test={len(test_files)}")

if __name__ == "__main__":
    create_splits()
