import os

import numpy as np
import torch
import torch.nn as nn
import torch.optim as optim
from torch.utils.data import DataLoader, Subset
from torch.cuda.amp import GradScaler, autocast
import timm
from sklearn.model_selection import StratifiedKFold
from sklearn.metrics import accuracy_score, precision_recall_fscore_support, f1_score
from torchvision.datasets import ImageFolder
import albumentations as A
from albumentations.pytorch import ToTensorV2
from PIL import Image
import csv
import argparse
import copy
from collections import Counter

# Set up paths
BASE_DIR = r"d:\COLLEGE\SEM-4\EDI-2\New folder\lung_cancer_detection"
TRAIN_DIR = os.path.join(BASE_DIR, "backend", "DataSet", "train")
OUTPUTS_DIR = os.path.join(BASE_DIR, "ml", "outputs")
os.makedirs(os.path.join(OUTPUTS_DIR, "reports"), exist_ok=True)
os.makedirs(os.path.join(OUTPUTS_DIR, "checkpoints"), exist_ok=True)

class AlbumentationImageFolder(ImageFolder):
    def __init__(self, root, transform=None):
        super().__init__(root)
        self.transform = transform
        
    def __getitem__(self, index):
        path, target = self.samples[index]
        image = Image.open(path).convert("RGB")
        image = np.array(image)
        if self.transform:
            augmented = self.transform(image=image)
            image = augmented['image']
        return image, target

def get_transforms(img_size):
    train_transform = A.Compose([
        A.Resize(img_size + 32, img_size + 32),
        A.RandomCrop(width=img_size, height=img_size),
        A.HorizontalFlip(p=0.5), # Medically acceptable mostly
        A.ShiftScaleRotate(shift_limit=0.05, scale_limit=0.05, rotate_limit=15, p=0.5),
        A.RandomBrightnessContrast(p=0.2),
        A.Normalize(mean=(0.485, 0.456, 0.406), std=(0.229, 0.224, 0.225)),
        ToTensorV2(),
    ])
    val_transform = A.Compose([
        A.Resize(img_size, img_size),
        A.Normalize(mean=(0.485, 0.456, 0.406), std=(0.229, 0.224, 0.225)),
        ToTensorV2(),
    ])
    return train_transform, val_transform

def get_class_weights(dataset):
    targets = dataset.targets
    class_counts = Counter(targets)
    total = len(targets)
    num_classes = len(class_counts)
    weights = [total / (num_classes * class_counts[i]) for i in range(num_classes)]
    return torch.FloatTensor(weights)

def train_epoch(model, loader, criterion, optimizer, scaler, device):
    model.train()
    running_loss = 0.0
    all_preds, all_labels = [], []
    
    for images, labels in loader:
        images, labels = images.to(device), labels.to(device)
        optimizer.zero_grad()
        
        with autocast():
            outputs = model(images)
            loss = criterion(outputs, labels)
            
        scaler.scale(loss).backward()
        scaler.step(optimizer)
        scaler.update()
        
        running_loss += loss.item() * images.size(0)
        preds = outputs.argmax(dim=1)
        all_preds.extend(preds.cpu().numpy())
        all_labels.extend(labels.cpu().numpy())
        
    epoch_loss = running_loss / len(loader.dataset)
    epoch_acc = accuracy_score(all_labels, all_preds)
    epoch_f1 = f1_score(all_labels, all_preds, average='macro')
    return epoch_loss, epoch_acc, epoch_f1

def evaluate(model, loader, criterion, device):
    model.eval()
    running_loss = 0.0
    all_preds, all_labels = [], []
    
    with torch.no_grad():
        for images, labels in loader:
            images, labels = images.to(device), labels.to(device)
            outputs = model(images)
            loss = criterion(outputs, labels)
            running_loss += loss.item() * images.size(0)
            preds = outputs.argmax(dim=1)
            all_preds.extend(preds.cpu().numpy())
            all_labels.extend(labels.cpu().numpy())
            
    epoch_loss = running_loss / len(loader.dataset)
    epoch_acc = accuracy_score(all_labels, all_preds)
    epoch_f1 = f1_score(all_labels, all_preds, average='macro', zero_division=0)
    return epoch_loss, epoch_acc, epoch_f1

def run_cv(args):
    device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
    print(f"Using device: {device} | Model: {args.model_name}")
    
    train_transform, val_transform = get_transforms(args.img_size)
    dataset_full = AlbumentationImageFolder(TRAIN_DIR) # Base wrapper
    targets = dataset_full.targets
    num_classes = len(dataset_full.classes)
    class_weights = get_class_weights(dataset_full).to(device)
    
    skf = StratifiedKFold(n_splits=args.folds, shuffle=True, random_state=42)
    fold_results = []
    
    for fold, (train_idx, val_idx) in enumerate(skf.split(np.zeros(len(targets)), targets)):
        print(f"\n--- Fold {fold+1}/{args.folds} ---")
        
        # Datasets & Loaders
        train_sub = Subset(dataset_full, train_idx)
        val_sub = Subset(dataset_full, val_idx)
        
        # Override transforms specifically for Subset wrapping (hack robust for Albumentations)
        class _SubsetWrapper(torch.utils.data.Dataset):
            def __init__(self, subset, transform):
                self.subset = subset
                self.transform = transform
            def __getitem__(self, index):
                path, target = self.subset.dataset.samples[self.subset.indices[index]]
                image = Image.open(path).convert("RGB")
                image = np.array(image)
                if self.transform:
                    image = self.transform(image=image)['image']
                return image, target
            def __len__(self):
                return len(self.subset)
                
        train_ds = _SubsetWrapper(train_sub, train_transform)
        val_ds = _SubsetWrapper(val_sub, val_transform)
        
        train_loader = DataLoader(train_ds, batch_size=args.batch_size, shuffle=True, num_workers=0)
        val_loader = DataLoader(val_ds, batch_size=args.batch_size, shuffle=False, num_workers=0)
        
        # Model
        model = timm.create_model(args.model_name, pretrained=True, num_classes=num_classes)
        model = model.to(device)
        
        criterion = nn.CrossEntropyLoss(weight=class_weights)
        optimizer = optim.AdamW(model.parameters(), lr=args.lr, weight_decay=1e-4)
        scheduler = optim.lr_scheduler.OneCycleLR(optimizer, max_lr=args.lr, steps_per_epoch=len(train_loader), epochs=args.epochs)
        scaler = GradScaler()
        
        best_f1 = 0.0
        best_state = None
        patience_counter = 0
        
        for epoch in range(1, args.epochs + 1):
            tr_loss, tr_acc, tr_f1 = train_epoch(model, train_loader, criterion, optimizer, scaler, device)
            val_loss, val_acc, val_f1 = evaluate(model, val_loader, criterion, device)
            scheduler.step()
            
            print(f"Epoch {epoch:02d} | Tr Loss: {tr_loss:.4f} F1: {tr_f1:.4f} | Val Loss: {val_loss:.4f} F1: {val_f1:.4f} Acc: {val_acc:.4f}")
            
            if val_f1 > best_f1:
                best_f1 = val_f1
                best_state = copy.deepcopy(model.state_dict())
                patience_counter = 0
            else:
                patience_counter += 1
                
            if patience_counter >= args.patience:
                print("Early stopping triggered.")
                break
                
        # Final evaluation for fold
        model.load_state_dict(best_state)
        val_loss, val_acc, val_f1 = evaluate(model, val_loader, criterion, device)
        fold_results.append((val_acc, val_f1))
        print(f"Fold {fold+1} Best Val Acc: {val_acc:.4f} | Val F1: {val_f1:.4f}")
        
        torch.save(best_state, os.path.join(OUTPUTS_DIR, "checkpoints", f"{args.model_name}_fold{fold}.pth"))
        
        if args.single_fold:
            print("Single fold flag passed, stopping after Fold 1.")
            break
            
    # Compile results
    accs = [f[0] for f in fold_results]
    f1s = [f[1] for f in fold_results]
    
    mean_acc = np.mean(accs)
    std_acc = np.std(accs)
    mean_f1 = np.mean(f1s)
    
    print(f"\n=== CV Results for {args.model_name} ===")
    print(f"Mean Accuracy: {mean_acc:.4f} ± {std_acc:.4f}")
    print(f"Mean Macro F1: {mean_f1:.4f}")
    
    res_csv = os.path.join(OUTPUTS_DIR, "reports", "experiment_summary.csv")
    file_exists = os.path.isfile(res_csv)
    with open(res_csv, mode='a', newline='') as f:
        writer = csv.writer(f)
        if not file_exists:
            writer.writerow(["model_name", "img_size", "epochs", "folds", "mean_acc", "std_acc", "mean_f1"])
        writer.writerow([args.model_name, args.img_size, args.epochs, args.folds if not args.single_fold else 1, round(mean_acc,4), round(std_acc,4), round(mean_f1,4)])

if __name__ == "__main__":
    parser = argparse.ArgumentParser()
    parser.add_argument("--model_name", type=str, default="resnet50")
    parser.add_argument("--img_size", type=int, default=224)
    parser.add_argument("--batch_size", type=int, default=32)
    parser.add_argument("--epochs", type=int, default=15)
    parser.add_argument("--folds", type=int, default=5)
    parser.add_argument("--lr", type=float, default=1e-3)
    parser.add_argument("--patience", type=int, default=4)
    parser.add_argument("--single_fold", action="store_true")
    args = parser.parse_args()
    
    run_cv(args)
