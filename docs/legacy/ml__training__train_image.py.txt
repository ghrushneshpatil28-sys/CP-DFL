"""
train_image.py  —  ResNet18 transfer learning for lung cancer image classification.

Model:    ResNet18 (pretrained ImageNet) — classifier head replaced
Strategy: Extensively rewritten to automatically detect dataset classes, freeze backbone, 
          and train for dataset-driven subtypes.
          
Run:
    python ml/training/train_image.py

Outputs:
    ml/saved_models/resnet18_model.pth
    ml/saved_models/class_mapping.json
    ml/saved_models/model_metadata.json
    ml/outputs/plots/training_curve.png
"""

import os
import sys
import json
import time
import copy
from collections import defaultdict, Counter

sys.path.insert(0, os.path.join(os.path.dirname(__file__), "..", ".."))

import torch
import torch.nn as nn
import torch.optim as optim
from torch.utils.data import Dataset, DataLoader
from torchvision import models, transforms
from PIL import Image

try:
    from sklearn.metrics import classification_report, confusion_matrix
    HAS_SKLEARN = True
except ImportError:
    HAS_SKLEARN = False

from ml.utils.config import (
    TRAIN_DIR, VALID_DIR, TEST_DIR,
    IMAGE_MODEL_PATH, CLASS_MAP_JSON, MODEL_META_JSON,
    IMAGE_SIZE, BATCH_SIZE, NUM_EPOCHS,
    LR_HEAD, LR_FINETUNE, EARLY_STOP_PAT, RANDOM_SEED,
    PLOTS_DIR, SAVED_MODELS_DIR, REPORTS_DIR
)

torch.manual_seed(RANDOM_SEED)

VALID_EXTENSIONS = {".jpg", ".jpeg", ".png", ".bmp", ".tiff", ".tif"}

# ─── Dynamic Dataset Discovery ────────────────────────────────────────────────

def discover_classes(base_dir: str):
    """Scan training directory to organically discover class names."""
    if not os.path.isdir(base_dir):
        return []
    
    # Extract subfolders that have images
    found_classes = []
    for d in sorted(os.listdir(base_dir)):
        d_path = os.path.join(base_dir, d)
        if os.path.isdir(d_path):
            found_classes.append(d)
    return found_classes

def format_class_name(raw_name: str):
    """Make raw folder names into presentable class names."""
    # Custom rule for known messy names, can scale implicitly
    if "squamous" in raw_name.lower():
        return "Squamous Cell Carcinoma"
    elif "adenocarcinoma" in raw_name.lower():
        return "Adenocarcinoma"
    elif "large.cell" in raw_name.lower() or "large_cell" in raw_name.lower():
        return "Large Cell Carcinoma"
    elif "normal" in raw_name.lower() or "no_cancer" in raw_name.lower():
        return "No Cancer"
    
    # Generic format
    return raw_name.replace("_", " ").replace(".", " ").title()

# ─── Dataset ───────────────────────────────────────────────────────────────────

class LungDatasetDynamic(Dataset):
    """Reads images from a split directory and maps dynamically."""

    MEAN = [0.485, 0.456, 0.406]
    STD  = [0.229, 0.224, 0.225]

    def __init__(self, split_dir: str, class_to_idx: dict, augment: bool = False):
        self.samples = []   # [(abs_path, class_int)]
        self.class_to_idx = class_to_idx
        
        # Build inverted lookup for prefix matching if exact match fails
        self.formatted_to_idx = {format_class_name(c).lower(): i for c, i in class_to_idx.items()}

        for folder in sorted(os.listdir(split_dir)):
            folder_path = os.path.join(split_dir, folder)
            if not os.path.isdir(folder_path):
                continue
            
            # Match folder to class ID
            cls_name = format_class_name(folder)
            if folder in self.class_to_idx:
                class_idx = self.class_to_idx[folder]
            elif cls_name.lower() in self.formatted_to_idx:
                class_idx = self.formatted_to_idx[cls_name.lower()]
            else:
                print(f"  [WARN] Unrecognised folder, skipping: {folder}")
                continue

            for fname in os.listdir(folder_path):
                if os.path.splitext(fname)[1].lower() not in VALID_EXTENSIONS:
                    continue
                fpath = os.path.join(folder_path, fname)
                self.samples.append((fpath, class_idx))

        if augment:
            self.transform = transforms.Compose([
                transforms.Resize((IMAGE_SIZE + 24, IMAGE_SIZE + 24)),
                transforms.RandomCrop(IMAGE_SIZE),
                transforms.RandomHorizontalFlip(),
                transforms.RandomRotation(10),
                transforms.ColorJitter(brightness=0.15, contrast=0.15),
                transforms.ToTensor(),
                transforms.Normalize(self.MEAN, self.STD),
            ])
        else:
            self.transform = transforms.Compose([
                transforms.Resize((IMAGE_SIZE, IMAGE_SIZE)),
                transforms.ToTensor(),
                transforms.Normalize(self.MEAN, self.STD),
            ])

    def __len__(self):
        return len(self.samples)

    def __getitem__(self, idx):
        fpath, label = self.samples[idx]
        try:
            img = Image.open(fpath).convert("RGB")
        except Exception:
            img = Image.new("RGB", (IMAGE_SIZE, IMAGE_SIZE))
        return self.transform(img), label

# ─── Model ────────────────────────────────────────────────────────────────────

def build_model(num_classes: int, device: torch.device):
    """Load pretrained ResNet18, freeze backbone, replace head."""
    model = models.resnet18(weights=models.ResNet18_Weights.IMAGENET1K_V1)

    # Freeze all backbone layers
    for param in model.parameters():
        param.requires_grad = False

    # Replace classifier head for ResNet18
    in_features = model.fc.in_features
    # We unfreeze the last layer automatically by replacing it
    model.fc = nn.Sequential(
        nn.Dropout(0.3),
        nn.Linear(in_features, 512),
        nn.BatchNorm1d(512),
        nn.ReLU(inplace=True),
        nn.Dropout(0.2),
        nn.Linear(512, num_classes),
    )
    return model.to(device)


def unfreeze_last_n_blocks(model, n: int = 1):
    """Unfreeze the last n feature blocks for fine-tuning. ResNet layer4 is target."""
    for param in model.layer4.parameters():
        param.requires_grad = True
    print(f"  [Finetune] Unfroze ResNet18 layer4 block.")

# ─── Training helpers ──────────────────────────────────────────────────────────

def train_one_epoch(model, loader, criterion, optimizer, device):
    model.train()
    total_loss, correct, total = 0.0, 0, 0
    for images, labels in loader:
        images, labels = images.to(device), labels.to(device)
        optimizer.zero_grad()
        outputs = model(images)
        loss = criterion(outputs, labels)
        loss.backward()
        optimizer.step()
        
        total_loss += loss.item() * images.size(0)
        correct += (outputs.argmax(1) == labels).sum().item()
        total   += images.size(0)
    return total_loss / max(total, 1), correct / max(total, 1)


@torch.no_grad()
def evaluate(model, loader, criterion, device):
    model.eval()
    total_loss, correct, total = 0.0, 0, 0
    all_preds = []
    all_labels = []
    
    for images, labels in loader:
        images, labels = images.to(device), labels.to(device)
        outputs = model(images)
        loss = criterion(outputs, labels)
        
        total_loss += loss.item() * images.size(0)
        preds = outputs.argmax(1)
        correct += (preds == labels).sum().item()
        total   += images.size(0)
        
        all_preds.extend(preds.cpu().tolist())
        all_labels.extend(labels.cpu().tolist())
        
    return total_loss / max(total, 1), correct / max(total, 1), all_preds, all_labels

def save_plot(history: dict, out_path: str):
    try:
        import matplotlib
        matplotlib.use("Agg")
        import matplotlib.pyplot as plt

        epochs = range(1, len(history["train_loss"]) + 1)
        fig, axes = plt.subplots(1, 2, figsize=(12, 4))

        axes[0].plot(epochs, history["train_loss"], label="Train Loss")
        axes[0].plot(epochs, history["val_loss"],   label="Val Loss")
        axes[0].set_title("Loss (ResNet18)")
        axes[0].set_xlabel("Epoch")
        axes[0].legend()

        axes[1].plot(epochs, history["train_acc"], label="Train Acc")
        axes[1].plot(epochs, history["val_acc"],   label="Val Acc")
        axes[1].set_title("Accuracy")
        axes[1].set_xlabel("Epoch")
        axes[1].legend()

        plt.tight_layout()
        plt.savefig(out_path, dpi=100)
        plt.close()
        print(f"  Training curve saved → {out_path}")
    except Exception as e:
        print(f"  [WARN] Could not save plot: {e}")

# ─── Main ──────────────────────────────────────────────────────────────────────

def main():
    print("=" * 60)
    print("  Lung Cancer Subtype Training  (ResNet18)")
    print("=" * 60)

    for d, name in [(TRAIN_DIR, "train"), (VALID_DIR, "valid")]:
        if not os.path.isdir(d):
            print(f"[ERROR] {name} directory not found: {d}")
            sys.exit(1)

    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    
    # ── Dynamic Classes Generation ──
    raw_classes = discover_classes(TRAIN_DIR)
    if not raw_classes:
        print(f"[ERROR] No classes found in {TRAIN_DIR}. Please check dataset.")
        sys.exit(1)

    # Build Class mapping mapping
    formatted_class_names = [format_class_name(c) for c in raw_classes]
    class_to_idx = {raw: i for i, raw in enumerate(raw_classes)}
    idx_to_class = {i: fmt for i, fmt in enumerate(formatted_class_names)}
    
    print(f"  Device       : {device}")
    print(f"  Found Classes: {formatted_class_names}")
    print(f"  Batch size   : {BATCH_SIZE}")
    print(f"  Image size   : {IMAGE_SIZE}x{IMAGE_SIZE}")

    # ── Datasets & loaders ────────────────────────────────────────────────────
    train_ds = LungDatasetDynamic(TRAIN_DIR, class_to_idx, augment=True)
    val_ds   = LungDatasetDynamic(VALID_DIR, class_to_idx, augment=False)

    print(f"  Train images : {len(train_ds)}")
    print(f"  Val images   : {len(val_ds)}")

    train_loader = DataLoader(train_ds, batch_size=BATCH_SIZE, shuffle=True, num_workers=0)
    val_loader   = DataLoader(val_ds,   batch_size=BATCH_SIZE, shuffle=False, num_workers=0)

    num_classes = len(formatted_class_names)
    model = build_model(num_classes, device)
    criterion = nn.CrossEntropyLoss()

    # ── Phase 1: Train head only (backbone frozen) ────────────────────────────
    print("\n  Phase 1: Training classifier head (backbone frozen) ...")
    optimizer = optim.Adam(
        filter(lambda p: p.requires_grad, model.parameters()),
        lr=LR_HEAD
    )
    scheduler = optim.lr_scheduler.ReduceLROnPlateau(optimizer, mode="min", factor=0.5, patience=2)

    best_val_loss = float("inf")
    best_val_acc  = 0.0
    best_state    = None
    patience_ctr  = 0
    history       = {"train_loss": [], "val_loss": [], "train_acc": [], "val_acc": []}
    best_val_preds, best_val_labels = [], []

    for epoch in range(1, NUM_EPOCHS + 1):
        t0 = time.time()
        tr_loss, tr_acc = train_one_epoch(model, train_loader, criterion, optimizer, device)
        va_loss, va_acc, va_preds, va_labels = evaluate(model, val_loader, criterion, device)
        scheduler.step(va_loss)

        history["train_loss"].append(tr_loss)
        history["val_loss"].append(va_loss)
        history["train_acc"].append(tr_acc)
        history["val_acc"].append(va_acc)

        elapsed = time.time() - t0
        print(f"  Epoch {epoch:02d}/{NUM_EPOCHS}  "
              f"train_loss={tr_loss:.4f}  train_acc={tr_acc:.3f}  "
              f"val_loss={va_loss:.4f}  val_acc={va_acc:.3f}  ({elapsed:.1f}s)")

        if va_loss < best_val_loss - 1e-4:
            best_val_loss = va_loss
            best_val_acc  = va_acc
            best_val_preds = va_preds
            best_val_labels = va_labels
            best_state    = copy.deepcopy(model.state_dict())
            patience_ctr  = 0
        else:
            patience_ctr += 1
            if patience_ctr >= EARLY_STOP_PAT:
                print(f"  [EarlyStop] No improvement for {EARLY_STOP_PAT} epochs. Stopping Phase 1.")
                break

    # ── Phase 2: Fine-tune last block ──────────────────────────────────────
    print("\n  Phase 2: Fine-tuning ResNet layer4 ...")
    model.load_state_dict(best_state)
    unfreeze_last_n_blocks(model, n=1)

    optimizer_ft = optim.Adam(
        filter(lambda p: p.requires_grad, model.parameters()),
        lr=LR_FINETUNE
    )
    scheduler_ft = optim.lr_scheduler.ReduceLROnPlateau(optimizer_ft, mode="min", factor=0.5, patience=2)

    patience_ctr = 0
    ft_epochs    = 5

    for epoch in range(1, ft_epochs + 1):
        t0 = time.time()
        tr_loss, tr_acc = train_one_epoch(model, train_loader, criterion, optimizer_ft, device)
        va_loss, va_acc, va_preds, va_labels = evaluate(model, val_loader, criterion, device)
        scheduler_ft.step(va_loss)

        history["train_loss"].append(tr_loss)
        history["val_loss"].append(va_loss)
        history["train_acc"].append(tr_acc)
        history["val_acc"].append(va_acc)

        elapsed = time.time() - t0
        print(f"  FT Epoch {epoch:02d}/{ft_epochs}  "
              f"train_loss={tr_loss:.4f}  train_acc={tr_acc:.3f}  "
              f"val_loss={va_loss:.4f}  val_acc={va_acc:.3f}  ({elapsed:.1f}s)")

        if va_loss < best_val_loss - 1e-4:
            best_val_loss = va_loss
            best_val_acc  = va_acc
            best_val_preds = va_preds
            best_val_labels = va_labels
            best_state    = copy.deepcopy(model.state_dict())
            patience_ctr  = 0
        else:
            patience_ctr += 1
            if patience_ctr >= EARLY_STOP_PAT:
                print("  [EarlyStop] Fine-tuning stopped early.")
                break

    # ── Final Evaluation Metrics (sklearn) ──────────────────────────────────────
    if HAS_SKLEARN:
        print("\n  [Metrics] Validation Output:")
        labels_ordered = [idx_to_class[i] for i in range(num_classes)]
        rep = classification_report(best_val_labels, best_val_preds, target_names=labels_ordered, zero_division=0)
        print(rep)
        
        cm = confusion_matrix(best_val_labels, best_val_preds)
        print("  Confusion Matrix:")
        print(cm)
        
        # Save metrics
        os.makedirs(REPORTS_DIR, exist_ok=True)
        with open(os.path.join(REPORTS_DIR, "classification_report.txt"), "w") as f:
            f.write(rep + "\n\nConfusion Matrix:\n" + str(cm))

    # ── Save best model ────────────────────────────────────────────────────────
    os.makedirs(SAVED_MODELS_DIR, exist_ok=True)
    model.load_state_dict(best_state)

    checkpoint = {
        "model_state_dict": best_state,
        "class_names":      formatted_class_names,
        "class_to_idx":     class_to_idx,
        "idx_to_class":     idx_to_class,
        "num_classes":      num_classes,
        "image_size":       IMAGE_SIZE,
        "best_val_loss":    best_val_loss,
        "best_val_acc":     best_val_acc,
    }
    torch.save(checkpoint, IMAGE_MODEL_PATH)
    print(f"\n  Best model saved → {IMAGE_MODEL_PATH}")
    print(f"  Best val accuracy : {best_val_acc:.4f}")

    # ── Save dynamic class mapping JSON ────────────────────────────────────────
    class_map = {
        "class_names":    formatted_class_names,
        "class_to_idx":   class_to_idx,
        "idx_to_class":   idx_to_class,
        "num_classes":    num_classes,
    }
    with open(CLASS_MAP_JSON, "w") as f:
        json.dump(class_map, f, indent=2)
    print(f"  Class map saved  → {CLASS_MAP_JSON}")

    # ── Save model metadata ────────────────────────────────────────────────────
    meta = {
        "architecture":    "ResNet18",
        "pretrained_on":   "ImageNet",
        "num_classes":     num_classes,
        "class_names":     formatted_class_names,
        "image_size":      IMAGE_SIZE,
        "batch_size":      BATCH_SIZE,
        "best_val_accuracy": round(best_val_acc, 4),
        "best_val_loss":   round(best_val_loss, 4),
        "framework":       "PyTorch",
        "note":            "Dynamic subtype classification with ResNet18",
    }
    with open(MODEL_META_JSON, "w") as f:
        json.dump(meta, f, indent=2)
    print(f"  Metadata saved   → {MODEL_META_JSON}")

    # ── Save training curve ────────────────────────────────────────────────────
    os.makedirs(PLOTS_DIR, exist_ok=True)
    save_plot(history, os.path.join(PLOTS_DIR, "training_curve.png"))

    print("\n  Training complete!")
    print("=" * 60)
    print("  Next step: Start FastAPI backend → uvicorn backend.main:app --reload")


if __name__ == "__main__":
    main()
