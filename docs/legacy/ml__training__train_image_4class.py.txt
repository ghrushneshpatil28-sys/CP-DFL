"""
train_image_4class.py  —  MobileNetV2 transfer learning for 4-class lung cancer classification.

Classes:
  0 - No Cancer
  1 - Adenocarcinoma
  2 - Large Cell Carcinoma
  3 - Squamous Cell Carcinoma

Model:    MobileNetV2 (pretrained ImageNet) — 4-class head
Strategy: Phase 1 = freeze backbone, train head (NUM_EPOCHS)
          Phase 2 = unfreeze last 2 blocks, fine-tune (up to 5 more epochs)

Dataset:  Uses backend/DataSet/train, backend/DataSet/valid directly.
          No manifest CSV needed (already split by the dataset provider).

Run:
    python ml/training/train_image_4class.py

Outputs:
    ml/saved_models/image_model_best.pth
    ml/saved_models/class_mapping.json
    ml/saved_models/model_metadata.json
    ml/outputs/plots/training_curve.png
"""

import os
import sys
import json
import time
import copy

sys.path.insert(0, os.path.join(os.path.dirname(__file__), "..", ".."))

import torch
import torch.nn as nn
import torch.optim as optim
from torch.utils.data import Dataset, DataLoader
from torchvision import models, transforms
from PIL import Image

from ml.utils.config import (
    TRAIN_DIR, VALID_DIR, TEST_DIR,
    IMAGE_MODEL_PATH, CLASS_MAP_JSON, MODEL_META_JSON,
    CLASS_NAMES, FOLDER_TO_LABEL,
    IMAGE_SIZE, BATCH_SIZE, NUM_EPOCHS,
    LR_HEAD, LR_FINETUNE, EARLY_STOP_PAT, RANDOM_SEED,
    PLOTS_DIR, SAVED_MODELS_DIR,
)

torch.manual_seed(RANDOM_SEED)

VALID_EXTENSIONS = {".jpg", ".jpeg", ".png", ".bmp", ".tiff", ".tif"}


# ─── Dataset ───────────────────────────────────────────────────────────────────

def _resolve_label(folder_name: str):
    """Map folder name → class label using exact + prefix match."""
    if folder_name in FOLDER_TO_LABEL:
        return FOLDER_TO_LABEL[folder_name]
    for key, label in FOLDER_TO_LABEL.items():
        if folder_name.startswith(key) or key.startswith(folder_name):
            return label
    return None


class LungDataset4Class(Dataset):
    """Reads images from a split directory (train / valid / test)."""

    MEAN = [0.485, 0.456, 0.406]
    STD  = [0.229, 0.224, 0.225]

    def __init__(self, split_dir: str, augment: bool = False):
        self.samples = []   # [(abs_path, class_int)]
        self.class_to_idx = {name: i for i, name in enumerate(CLASS_NAMES)}
        skipped = 0

        for folder in sorted(os.listdir(split_dir)):
            folder_path = os.path.join(split_dir, folder)
            if not os.path.isdir(folder_path):
                continue

            label = _resolve_label(folder)
            if label is None:
                print(f"  [WARN] Unrecognised folder, skipping: {folder}")
                continue

            class_idx = self.class_to_idx[label]
            for fname in os.listdir(folder_path):
                if os.path.splitext(fname)[1].lower() not in VALID_EXTENSIONS:
                    continue
                fpath = os.path.join(folder_path, fname)
                self.samples.append((fpath, class_idx))

        if augment:
            self.transform = transforms.Compose([
                transforms.Resize((IMAGE_SIZE + 24, IMAGE_SIZE + 24)),
                transforms.RandomCrop(IMAGE_SIZE),
                transforms.RandomHorizontalFlip(),
                transforms.RandomRotation(10),
                transforms.ColorJitter(brightness=0.15, contrast=0.15),
                transforms.ToTensor(),
                transforms.Normalize(self.MEAN, self.STD),
            ])
        else:
            self.transform = transforms.Compose([
                transforms.Resize((IMAGE_SIZE, IMAGE_SIZE)),
                transforms.ToTensor(),
                transforms.Normalize(self.MEAN, self.STD),
            ])

    def __len__(self):
        return len(self.samples)

    def __getitem__(self, idx):
        fpath, label = self.samples[idx]
        try:
            img = Image.open(fpath).convert("RGB")
        except Exception:
            img = Image.new("RGB", (IMAGE_SIZE, IMAGE_SIZE))
        return self.transform(img), label


# ─── Model ────────────────────────────────────────────────────────────────────

def build_model(num_classes: int, device: torch.device):
    """Load pretrained MobileNetV2, freeze backbone, replace head."""
    model = models.mobilenet_v2(weights=models.MobileNet_V2_Weights.IMAGENET1K_V1)

    # Freeze all backbone layers
    for param in model.features.parameters():
        param.requires_grad = False

    # Replace classifier head
    in_features = model.classifier[1].in_features
    model.classifier = nn.Sequential(
        nn.Dropout(0.3),
        nn.Linear(in_features, 512),
        nn.BatchNorm1d(512),
        nn.ReLU(inplace=True),
        nn.Dropout(0.2),
        nn.Linear(512, num_classes),
    )
    return model.to(device)


def unfreeze_last_n_blocks(model, n: int = 2):
    """Unfreeze the last n feature blocks for fine-tuning."""
    features = list(model.features.children())
    for block in features[-n:]:
        for param in block.parameters():
            param.requires_grad = True
    print(f"  [Finetune] Unfroze last {n} MobileNetV2 feature blocks.")


# ─── Training helpers ──────────────────────────────────────────────────────────

def train_one_epoch(model, loader, criterion, optimizer, device):
    model.train()
    total_loss, correct, total = 0.0, 0, 0
    for images, labels in loader:
        images, labels = images.to(device), labels.to(device)
        optimizer.zero_grad()
        outputs = model(images)
        loss = criterion(outputs, labels)
        loss.backward()
        optimizer.step()
        total_loss += loss.item() * images.size(0)
        correct += (outputs.argmax(1) == labels).sum().item()
        total   += images.size(0)
    return total_loss / max(total, 1), correct / max(total, 1)


@torch.no_grad()
def evaluate(model, loader, criterion, device):
    model.eval()
    total_loss, correct, total = 0.0, 0, 0
    for images, labels in loader:
        images, labels = images.to(device), labels.to(device)
        outputs = model(images)
        loss = criterion(outputs, labels)
        total_loss += loss.item() * images.size(0)
        correct += (outputs.argmax(1) == labels).sum().item()
        total   += images.size(0)
    return total_loss / max(total, 1), correct / max(total, 1)


def run_test_evaluation(model, test_dir: str, device: torch.device):
    """Evaluate on the test split and print per-class accuracy."""
    import collections

    test_ds = LungDataset4Class(test_dir, augment=False)
    if len(test_ds) == 0:
        print("  [WARN] Test dataset is empty, skipping test evaluation.")
        return

    criterion = nn.CrossEntropyLoss()
    test_loader = DataLoader(test_ds, batch_size=BATCH_SIZE, shuffle=False, num_workers=0)
    test_loss, test_acc = evaluate(model, test_loader, criterion, device)

    print(f"\n  Test Loss: {test_loss:.4f}  |  Test Accuracy: {test_acc:.4f}")

    # Per-class accuracy
    model.eval()
    per_class_correct = collections.defaultdict(int)
    per_class_total   = collections.defaultdict(int)

    with torch.no_grad():
        for images, labels in test_loader:
            images, labels = images.to(device), labels.to(device)
            preds = model(images).argmax(1)
            for pred, gt in zip(preds.tolist(), labels.tolist()):
                per_class_total[gt] += 1
                if pred == gt:
                    per_class_correct[gt] += 1

    print("  Per-class test accuracy:")
    for idx, cls in enumerate(CLASS_NAMES):
        tot = per_class_total.get(idx, 0)
        acc = per_class_correct.get(idx, 0) / max(tot, 1)
        print(f"    {cls:30s}: {acc:.4f}  ({per_class_correct.get(idx,0)}/{tot})")

    return test_acc


def save_plot(history: dict, out_path: str):
    try:
        import matplotlib
        matplotlib.use("Agg")
        import matplotlib.pyplot as plt

        epochs = range(1, len(history["train_loss"]) + 1)
        fig, axes = plt.subplots(1, 2, figsize=(12, 4))

        axes[0].plot(epochs, history["train_loss"], label="Train Loss")
        axes[0].plot(epochs, history["val_loss"],   label="Val Loss")
        axes[0].set_title("Loss  (4-class MobileNetV2)")
        axes[0].set_xlabel("Epoch")
        axes[0].legend()

        axes[1].plot(epochs, history["train_acc"], label="Train Acc")
        axes[1].plot(epochs, history["val_acc"],   label="Val Acc")
        axes[1].set_title("Accuracy")
        axes[1].set_xlabel("Epoch")
        axes[1].legend()

        plt.tight_layout()
        plt.savefig(out_path, dpi=100)
        plt.close()
        print(f"  Training curve saved → {out_path}")
    except Exception as e:
        print(f"  [WARN] Could not save plot: {e}")


# ─── Main ──────────────────────────────────────────────────────────────────────

def main():
    print("=" * 60)
    print("  4-Class Lung Cancer Image Training  (MobileNetV2)")
    print("=" * 60)

    for d, name in [(TRAIN_DIR, "train"), (VALID_DIR, "valid")]:
        if not os.path.isdir(d):
            print(f"[ERROR] {name} directory not found: {d}")
            sys.exit(1)

    device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
    print(f"  Device       : {device}")
    print(f"  Classes      : {CLASS_NAMES}")
    print(f"  Batch size   : {BATCH_SIZE}")
    print(f"  Image size   : {IMAGE_SIZE}x{IMAGE_SIZE}")

    # ── Datasets & loaders ────────────────────────────────────────────────────
    train_ds = LungDataset4Class(TRAIN_DIR, augment=True)
    val_ds   = LungDataset4Class(VALID_DIR, augment=False)

    print(f"  Train images : {len(train_ds)}")
    print(f"  Val images   : {len(val_ds)}")

    # Cross-verify class counts
    from collections import Counter
    tr_dist = Counter(label for _, label in train_ds.samples)
    print("  Train class distribution:")
    for i, cls in enumerate(CLASS_NAMES):
        print(f"    [{i}] {cls}: {tr_dist.get(i, 0)}")
    print()

    train_loader = DataLoader(train_ds, batch_size=BATCH_SIZE, shuffle=True, num_workers=0)
    val_loader   = DataLoader(val_ds,   batch_size=BATCH_SIZE, shuffle=False, num_workers=0)

    num_classes = len(CLASS_NAMES)
    model = build_model(num_classes, device)
    criterion = nn.CrossEntropyLoss()

    # ── Phase 1: Train head only (backbone frozen) ────────────────────────────
    print("  Phase 1: Training classifier head (backbone frozen) ...")
    optimizer = optim.Adam(
        filter(lambda p: p.requires_grad, model.parameters()),
        lr=LR_HEAD
    )
    scheduler = optim.lr_scheduler.ReduceLROnPlateau(optimizer, mode="min", factor=0.5, patience=2)

    best_val_loss = float("inf")
    best_val_acc  = 0.0
    best_state    = None
    patience_ctr  = 0
    history       = {"train_loss": [], "val_loss": [], "train_acc": [], "val_acc": []}

    for epoch in range(1, NUM_EPOCHS + 1):
        t0 = time.time()
        tr_loss, tr_acc = train_one_epoch(model, train_loader, criterion, optimizer, device)
        va_loss, va_acc = evaluate(model, val_loader, criterion, device)
        scheduler.step(va_loss)

        history["train_loss"].append(tr_loss)
        history["val_loss"].append(va_loss)
        history["train_acc"].append(tr_acc)
        history["val_acc"].append(va_acc)

        elapsed = time.time() - t0
        print(f"  Epoch {epoch:02d}/{NUM_EPOCHS}  "
              f"train_loss={tr_loss:.4f}  train_acc={tr_acc:.3f}  "
              f"val_loss={va_loss:.4f}  val_acc={va_acc:.3f}  ({elapsed:.1f}s)")

        if va_loss < best_val_loss - 1e-4:
            best_val_loss = va_loss
            best_val_acc  = va_acc
            best_state    = copy.deepcopy(model.state_dict())
            patience_ctr  = 0
        else:
            patience_ctr += 1
            if patience_ctr >= EARLY_STOP_PAT:
                print(f"  [EarlyStop] No improvement for {EARLY_STOP_PAT} epochs. Stopping Phase 1.")
                break

    # ── Phase 2: Fine-tune last 2 blocks ──────────────────────────────────────
    print()
    print("  Phase 2: Fine-tuning last 2 MobileNetV2 blocks ...")
    model.load_state_dict(best_state)
    unfreeze_last_n_blocks(model, n=2)

    optimizer_ft = optim.Adam(
        filter(lambda p: p.requires_grad, model.parameters()),
        lr=LR_FINETUNE
    )
    scheduler_ft = optim.lr_scheduler.ReduceLROnPlateau(optimizer_ft, mode="min", factor=0.5, patience=2)

    patience_ctr = 0
    ft_epochs    = 5

    for epoch in range(1, ft_epochs + 1):
        t0 = time.time()
        tr_loss, tr_acc = train_one_epoch(model, train_loader, criterion, optimizer_ft, device)
        va_loss, va_acc = evaluate(model, val_loader, criterion, device)
        scheduler_ft.step(va_loss)

        history["train_loss"].append(tr_loss)
        history["val_loss"].append(va_loss)
        history["train_acc"].append(tr_acc)
        history["val_acc"].append(va_acc)

        elapsed = time.time() - t0
        print(f"  FT Epoch {epoch:02d}/{ft_epochs}  "
              f"train_loss={tr_loss:.4f}  train_acc={tr_acc:.3f}  "
              f"val_loss={va_loss:.4f}  val_acc={va_acc:.3f}  ({elapsed:.1f}s)")

        if va_loss < best_val_loss - 1e-4:
            best_val_loss = va_loss
            best_val_acc  = va_acc
            best_state    = copy.deepcopy(model.state_dict())
            patience_ctr  = 0
        else:
            patience_ctr += 1
            if patience_ctr >= EARLY_STOP_PAT:
                print("  [EarlyStop] Fine-tuning stopped early.")
                break

    # ── Save best model ────────────────────────────────────────────────────────
    os.makedirs(SAVED_MODELS_DIR, exist_ok=True)
    model.load_state_dict(best_state)

    checkpoint = {
        "model_state_dict": best_state,
        "class_names":      CLASS_NAMES,
        "num_classes":      num_classes,
        "image_size":       IMAGE_SIZE,
        "best_val_loss":    best_val_loss,
        "best_val_acc":     best_val_acc,
    }
    torch.save(checkpoint, IMAGE_MODEL_PATH)
    print(f"\n  Best model saved → {IMAGE_MODEL_PATH}")
    print(f"  Best val accuracy : {best_val_acc:.4f}")

    # ── Save class mapping JSON ────────────────────────────────────────────────
    class_map = {
        "class_names":    CLASS_NAMES,
        "class_to_idx":   {name: i for i, name in enumerate(CLASS_NAMES)},
        "idx_to_class":   {i: name for i, name in enumerate(CLASS_NAMES)},
        "folder_to_label": FOLDER_TO_LABEL,
        "num_classes":    num_classes,
    }
    with open(CLASS_MAP_JSON, "w") as f:
        json.dump(class_map, f, indent=2)
    print(f"  Class map saved  → {CLASS_MAP_JSON}")

    # ── Save model metadata ────────────────────────────────────────────────────
    meta = {
        "architecture":    "MobileNetV2",
        "pretrained_on":   "ImageNet",
        "num_classes":     num_classes,
        "class_names":     CLASS_NAMES,
        "image_size":      IMAGE_SIZE,
        "batch_size":      BATCH_SIZE,
        "best_val_accuracy": round(best_val_acc, 4),
        "best_val_loss":   round(best_val_loss, 4),
        "framework":       "PyTorch",
        "note":            "4-class: No Cancer / Adenocarcinoma / Large Cell Carcinoma / Squamous Cell Carcinoma",
    }
    with open(MODEL_META_JSON, "w") as f:
        json.dump(meta, f, indent=2)
    print(f"  Metadata saved   → {MODEL_META_JSON}")

    # ── Save training curve ────────────────────────────────────────────────────
    os.makedirs(PLOTS_DIR, exist_ok=True)
    save_plot(history, os.path.join(PLOTS_DIR, "training_curve_4class.png"))

    # ── Test evaluation ────────────────────────────────────────────────────────
    if os.path.isdir(TEST_DIR):
        print("\n  Running test-set evaluation ...")
        run_test_evaluation(model, TEST_DIR, device)

    print("\n  Training complete!")
    print("=" * 60)
    print("  Next step: Start FastAPI backend → uvicorn backend.main:app --reload")


if __name__ == "__main__":
    main()
