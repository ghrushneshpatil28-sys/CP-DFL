# Let's write a dedicated pool merger.
import os, glob

BASE_DIR = r"d:\COLLEGE\SEM-4\EDI-2\New folder\lung_cancer_detection\backend\DataSet"

def unify_dataset():
    pool_dir = os.path.join(BASE_DIR, "pool")
    os.makedirs(pool_dir, exist_ok=True)
    
    # 1. Move everything into pool
    for split in ["train", "valid", "test"]:
        split_dir = os.path.join(BASE_DIR, split)
        if not os.path.exists(split_dir): continue
        
        for class_name in os.listdir(split_dir):
            class_dir = os.path.join(split_dir, class_name)
            if not os.path.isdir(class_dir): continue
            
            target_class_dir = os.path.join(pool_dir, class_name)
            os.makedirs(target_class_dir, exist_ok=True)
            
            for f in os.listdir(class_dir):
                src = os.path.join(class_dir, f)
                # Assign unique ID to prevent overlap
                dst = os.path.join(target_class_dir, f"{split}_{f}")
                shutil.move(src, dst)
        
        # safely remove empty split dir
        try:
            shutil.rmtree(split_dir)
        except Exception:
            pass

    print("Unified all datasets into DataSet/pool")

if __name__ == "__main__":
    unify_dataset()
